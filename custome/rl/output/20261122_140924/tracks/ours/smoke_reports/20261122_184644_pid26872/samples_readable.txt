def mod_fibonacci_sum(a, b, n, p=998244353):
    """
    Computes the sum of the first n Fibonacci numbers generated using modular arithmetic,
    where each term is computed as fib(n) = (fib(n-1) + fib(n-2)) % p for n >= 2,
    starting from fib(0) = a and fib(1) = b.
    
    Parameters:
    - a: A positive integer representing the first term of the sequence.
    - b: A positive integer representing the second term of the sequence.
    - n: An integer denoting the number of terms to sum up.
    - p: The modulus for the operation (default is 998244353).
    
    Returns:
    - The sum of the first n terms of the modified Fibonacci sequence modulo p.
    """

    # Base cases: if either a or b is 0, then all Fibonacci numbers will also be 0
    if a == 0 and b == 0:
        return 0
    
    # Initial conditions: fib(0) = a, fib(1) = b
    if n == 1:
        return a
    
    fib_sequence = [a, b]

    # Compute the next terms until we reach enough terms to cover 'n'
    while len(fib_sequence) < n:
        new_term = (fib_sequence[-1] + fib_sequence[-2]) % p
        fib_sequence.append(new_term)

    # Sum the terms and return modulo p
    result_sum = sum(fib_sequence[:n])
    return result_sum % p

def twisted_fibonacci(n: int) -> int:
    """
    Calculate the nth term of a modified Fibonacci sequence.
    
    Args:
        n (int): Position in the sequence (0-indexed).
        
    Returns:
        int: The value of the n-th term in the modified Fibonacci sequence.

        - Return -1 for n < 0.
        - Return 0 for n == 0.
        - Return 1 for n == 1.

        f(0) = 0, f(1) = 1
        
    Formula:
        f(n) = {
            3*f(n-1) + 2*f(n-2), if n is odd,
            2*f(n-1) + 3*f(n-2), if n is even
        }
    """
    
    # Base case checks
    if n < 0:
        return -1
    
    elif n == 0:
        return 0
    
    elif n == 1:
        return 1
    
    else:
        # Initialize the first two terms
        prev_two_terms = [0, 1]
        
        # Loop from 2 to n
        for i in range(2, n+1):
            # Calculate the next term based on the parity of n
            if i % 2 == 0:
                next_term = 3*prev_two_terms[0] + 2*prev_two_terms[1]

            # Print intermediate values to debug
            print(f"At step {i}, fib_{n} = {next_term}")
            
            # Return the current term
            return next_term
