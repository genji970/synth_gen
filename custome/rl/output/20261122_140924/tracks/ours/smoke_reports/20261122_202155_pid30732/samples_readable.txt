import math

def find_smallest_fib_sum(n: int, k: int) -> int:
    """
    Finds the smallest positive integer greater than 'k' that can be expressed as the sum of exactly 'n' distinct Fibonacci numbers.
    
    Parameters:
    n (int): Number of distinct Fibonacci numbers to use in the sum.
    k (int): Starting point for generating Fibonacci numbers, must be greater than k.
    
    Returns:
    int: The smallest possible value that equals the sum of n distinct Fibonacci numbers starting from k. Returns -1 if no such combination exists.
    
    Edge Cases:
    - No valid sums are possible if n <= 0 or k < 0.
    - Efficient computation avoids potential overflow errors due to large Fibonacci numbers.
    """

    # Check for invalid input
    if n <= 0 or k < 0:
        return -1
    
    # Initialize the first two Fibonacci numbers
    fib_0 = 0
    fib_1 = 1
    fibs = [fib_0, fib_1]

    # Generate Fibonacci numbers up to the nth term
    while len(fibs) < n:
        next_fib = fibs[-1] + fibs[-2]
        fibs.append(next_fib)

    # Calculate the minimum sum S(n)
    min_sum = float('inf')
    for m in range(k, fibs[-1]):
        current_sum = sum(fibs[0:m])
        if current_sum == min_sum and current_sum != 0:
            break
        elif current_sum > min_sum and current_sum % fibs[m] == 0:
            min_sum = current_sum
            
    return min_sum if min_sum != float('inf') else -1

# Helper function to calculate the nth Fibonacci number using an iterative approach
def fibonacci(n: int) -> int:
    # Base case initialization
    if n == 0:
        return 0
    elif n == 1:
        return 1
    
    # Initialize the first two Fibonacci numbers
    prev, curr = 0, 1
    
    # Iterate up to n to calculate the nth Fibonacci number
    for _ in range(n):
        temp = curr
        curr = prev + curr
        prev = temp
    
    return curr
