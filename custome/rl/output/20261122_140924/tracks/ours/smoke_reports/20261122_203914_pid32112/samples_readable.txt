    """
    Calculate the first 'n' terms of the modified Fibonacci sequence,
    where each term is multiplied by either 'a' or 'b'.
    If both 'a' and 'b' are not zero, include at least one term from each multiplication.

    :param a: A positive integer representing the multiplier for odd positions
    :param b: A positive integer representing the multiplier for even positions
    :param n: An integer indicating how many terms of the sequence to generate

    :return: A list containing the first 'n' terms of the modified Fibonacci sequence.
    """
    
    # Initialize variables
    fib = [0, 1]  # Start with the first two numbers in the Fibonacci sequence
    result = []  # Use an empty list to store the terms
    
    # Iterate through the desired number of terms
    while len(result) < n:
        next_term = a * fib[-2] + b * fib[-3]
        
        # Append current term to the result list
        result.append(next_term)
        
        # Update the previous three terms based on the calculated next term
        fib = [fib[1], next_term, fib[0]]
    
    return result
